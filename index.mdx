export const CopyCommand = ({ command }) => {
  const [copied, setCopied] = useState(false);

  const copy = () => {
    navigator.clipboard.writeText(command);
    setCopied(true);
    setTimeout(() => setCopied(false), 1600);
  };

  return (
    <button
      type="button"
      className="mint-home-copy"
      onClick={copy}
      aria-label={`Copy ${command} to clipboard`}
    >
      <span className="mint-home-copy-prompt" aria-hidden="true">
        $
      </span>
      <code className="mint-home-copy-command">{command}</code>
      <span className={copied ? "mint-home-copy-icon mint-home-copy-icon--done" : "mint-home-copy-icon"}>
        {copied ? (
          <Icon>
            <path d="M20 6 9 17l-5-5" />
          </Icon>
        ) : (
          <Icon>
            <rect width="14" height="14" x="8" y="8" rx="2" ry="2" />
            <path d="M4 16c-1.1 0-2-.9-2-2V4c0-1.1.9-2 2-2h10c1.1 0 2 .9 2 2" />
          </Icon>
        )}
      </span>
    </button>
  );
};
